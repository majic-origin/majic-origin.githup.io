### 💡 用处

1. 统一给不同的方法套上计时逻辑，避免重复写 start/end 代码；
2. 用匿名内部类实现接口，把「要执行的代码」和「计时逻辑」分开；
3. 实现方法执行耗时的通用统计，比如你写的筛法、排序算法都可以直接用。

### ✅ 优势

1. 代码复用：计时逻辑写一次，所有方法都能直接用；
2. 结构清晰：业务代码和计时代码分离，互不干扰；
3. 灵活扩展：想加新的方法，只要实现接口就行，不用改计时逻辑；
4. 避免重复：不用每个方法都写 System.currentTimeMillis()。

```

//
package time;

public class TimeCostDemo {

    // 1. 定义一个通用接口：代表一段要执行的任务
    public interface Task {
        void execute();
    }

    // 2. 通用计时工具方法
    public static void runWithTime(Task task) {
        long start = System.currentTimeMillis();
        // 执行传入的任务
        task.execute();
        long end = System.currentTimeMillis();
        System.out.println("方法执行耗时：" + (end - start) + " 毫秒");
    }

    // 示例方法：比如你的筛法求素数
    public static void primeSieveDemo() {
        int max = 1000000;
        boolean[] isPrime = new boolean[max + 1];
        for (int i = 2; i <= max; i++) {
            isPrime[i] = true;
        }
        for (int i = 2; i * i <= max; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= max; j += i) {
                    isPrime[j] = false;
                }
            }
        }
        int count = 0;
        for (int i = 2; i <= max; i++) {
            if (isPrime[i]) count++;
        }
        System.out.println("100万以内素数个数：" + count);
    }

    public static void main(String[] args) {
        // 3. 用匿名内部类实现 Task 接口，把方法传给计时工具
        runWithTime(new Task() {
            @Override
            public void execute() {
                primeSieveDemo(); // 要计时的业务方法
            }
        });

        // 也可以直接写别的方法
        runWithTime(new Task() {
            @Override
            public void execute() {
                // 随便写个循环，看看耗时
                long sum = 0;
                for (int i = 0; i < 100000000; i++) {
                    sum += i;
                }
                System.out.println("循环计算完成，sum=" + sum);
            }
        });
    }
}
```
